\begin{frame}{Four Real Incidents}
  \small
  \textbf{2015 \textbullet\ Jeep Cherokee, from the sofa.} Two researchers took over a Jeep on the highway from \textbf{16 km away}, in through its internet-facing infotainment unit, then pivoting to the CAN bus to work the engine, steering and brakes. No physical access, ever. {\scriptsize(\href{https://www.wired.com/2015/07/hackers-remotely-kill-jeep-highway/}{wired.com})}

  \vspace{1.2mm}
  \textbf{2016 \textbullet\ Tesla Model S, from kilometres away.} Tencent's Keen Security Lab chained the car's web browser into the CAN bus and hit the brakes of a \emph{moving} Model S remotely. Tesla pushed an over-the-air fix within ten days. {\scriptsize(\href{https://keenlab.tencent.com/en/2016/09/19/Keen-Security-Lab-of-Tencent-Car-Hacking-Research-Remote-Attack-to-Tesla-Cars/}{keenlab.tencent.com})}

  \vspace{1.2mm}
  \textbf{2022 \textbullet\ Kia \& Hyundai, the ``Kia Boys''.} Certain \textbf{US-market} Kia/Hyundai models (roughly 9 million, 2011--2022) shipped \textbf{without an engine immobilizer}. That design gap is separate from the theft \emph{method}: thieves forced the ignition and started the car with a USB-A plug, spread as the viral TikTok ``Kia Challenge''. {\scriptsize(\href{https://hbsslaw.com/press/hyundai-kia-car-theft-defect/hyundai-kia-theft-class-action-lawsuit-reaches-settlement-valued-at-more-than-200-million}{hbsslaw.com})}

  \vspace{1.2mm}
  \textbf{2024 \textbullet\ Any Kia, from a number plate.} Researchers abused Kia's dealer portal to control almost any Kia built after 2013 in about \alert{30 seconds}, using only its license plate: unlock, start, locate, and pull the owner's personal details. {\scriptsize(\href{https://samcurry.net/hacking-kia}{samcurry.net/hacking-kia})}
\end{frame}

\begin{frame}{The Consequences: a Bug Is Now a Recall}
  \small
  \setlength{\itemsep}{2pt}
  \begin{itemize}\setlength{\itemsep}{2pt}
    \item \textbf{Jeep (2015)}: \textbf{1.4 million} vehicles recalled, the industry's \emph{first} cybersecurity recall. A software flaw became a physical, logistical one.
    \item \textbf{Kia \& Hyundai}: \textbf{$\sim$9 million} cars exposed, a \textbf{\$200M} class-action settlement (up to \$145M to owners), free immobilizer software retrofits, and some US insurers refusing to cover the affected models.
    \item \textbf{Tesla (2016)}: fixed across the whole fleet by an \textbf{over-the-air update in 10 days}, the modern alternative to a recall, when you've engineered for it.
    \item \textbf{Dealerships (CDK, 2024)}: ransomware on one software vendor froze \textbf{$\sim$15,000} dealerships for weeks, an estimated \textbf{\$1 billion} in losses.
  \end{itemize}
  \begin{alertblock}{The throughline}
    In a car, a vulnerability isn't just a CVE. It's recalls, class actions, insurers walking away, and brand damage.
  \end{alertblock}
  {\scriptsize\color{tumgrey} Sources: Jeep recall, Chrysler/NHTSA (2015); Hyundai--Kia settlement, \href{https://hbsslaw.com/press/hyundai-kia-car-theft-defect/hyundai-kia-theft-class-action-lawsuit-reaches-settlement-valued-at-more-than-200-million}{Hagens Berman (2023)}; Tesla OTA fix, \href{https://keenlab.tencent.com/en/2016/09/19/Keen-Security-Lab-of-Tencent-Car-Hacking-Research-Remote-Attack-to-Tesla-Cars/}{Keen Security Lab (2016)}; CDK losses, \href{https://www.foxbusiness.com/markets/car-dealership-losses-from-cdk-software-outage-could-reach-1-billion-study-finds}{Anderson Economic Group (2024)}.}
\end{frame}

\begin{frame}{Automotive 2025, in Five Numbers}
  \begin{itemize}
    \item \alert{494} publicly reported incidents in 2025 (409 in 2024, 295 in 2023).
    \item \alert{44\%} were ransomware-related, roughly double 2024's share.
    \item \alert{92\%} were carried out remotely; \alert{86\%} needed no physical proximity to the vehicle.
    \item \alert{67\%} came in through telematics, cloud platforms or APIs, not the CAN bus.
    \item \alert{68\%} led to a data or privacy breach.
  \end{itemize}
  \vspace{1mm}
  {\scriptsize ``Publicly reported'' = incidents documented from public sources; each percentage is a share of those 494.}
  \source{Upstream Security, \emph{Global Automotive \& Smart Mobility Cybersecurity Report 2026} (8th annual; published Feb 2026, covering 2025): \href{https://upstream.auto/}{upstream.auto}}
\end{frame}
